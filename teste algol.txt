create TABLE clientes (
    CD_CLIENTES integer primary key,
    NM_CLIENTES varchar(100),
    CD_CIDADE integer,
    USUARIO varchar(100),
    foreign key (CD_CIDADE) references Cidades(CD_CIDADE) ON DELETE CASCADE ) ;

create table Cidades(
    CD_CIDADE integer primary key,
    NM_CIDADE varchar(100),
    USUARIO varchar(100)
);

create table Dependentes(
    CD_DEPENDENTE integer primary KEY,
    NM_DEPENDENTE varchar(100),
    CD_CLIENTES INTEGER,
    USUARIO varchar (100),
    foreign KEY (CD_CLIENTES) references CLIENTES(CD_CLIENTES ON DELETE CASCADE));

CREATE TABLE LOGS (
    CD_LOG integer primary KEY,
    DESCRICAO VARCHAR (100),
    DT_LOG date,
    HR_LOG timestamp);
